### Toast 系统（内嵌实现，非独立组件）
```typescript
const toasts = ref<{ id: number; message: string; type: 'info' | 'success' | 'error' }[]>([]);
function showToast(message: string, type = 'info') {
  const id = Date.now();
  toasts.value.push({ id, message, type });
  setTimeout(() => { toasts.value = toasts.value.filter(t => t.id !== id); }, 3000);
}
```
