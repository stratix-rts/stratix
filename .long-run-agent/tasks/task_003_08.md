## 实现方案

1. 在 `src/stratix-rts/zones/TaskZone.ts` 中：
```typescript
import { Building, BuildingFactory, BuildingConfig } from '../buildings/Building';
import { BuildingType } from '../buildings/types/BuildingType';

export class TaskZone extends BaseZone {
  private building?: Building;
  
  public createBuilding(buildingType: BuildingType): void {
    if (this.building) {
      return; // 已有建筑
    }
    
    const config: BuildingConfig = {
      x: this.x,
      y: this.y,
      width: this.zoneWidth,
      height: this.zoneHeight,
      buildingType
    };
    
    this.building = BuildingFactory.create(this.scene, config);
    this.add(this.building); // 加入 Container
    
    // 隐藏原有的 graphics 边框（可选）
    this.graphics.setVisible(false);
  }
  
  public getBuilding(): Building | undefined {
    return this.building;
  }
}
```

2. 在 UnifiedZoneManager 中集成：
```typescript
// 创建 Zone 时，如果指定了 buildingType，调用 createBuilding
const zone = new TaskZone(this.scene, config);
if (config.buildingType) {
  zone.createBuilding(config.buildingType);
}
```
