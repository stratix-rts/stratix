## 实现方案

1. 创建 `src/stratix-rts/buildings/RoomRenderer.ts`：
```typescript
export class RoomRenderer {
  private scene: Phaser.Scene;
  private roomGraphics: Phaser.GameObjects.Graphics;
  
  constructor(scene: Phaser.Scene) {
    this.scene = scene;
    this.roomGraphics = scene.add.graphics();
  }
  
  public renderRooms(rooms: Room[]): void {
    this.roomGraphics.clear();
    
    for (const room of rooms) {
      const config = room.getConfig();
      const color = config.state === RoomState.Occupied ? 0x22c55e : 0x9ca3af;
      
      this.roomGraphics.fillStyle(color, 0.8);
      this.roomGraphics.fillRect(
        config.x - config.width/2,
        config.y - config.height/2,
        config.width,
        config.height
      );
      
      // 边框
      this.roomGraphics.lineStyle(1, 0x000000, 0.3);
      this.roomGraphics.strokeRect(
        config.x - config.width/2,
        config.y - config.height/2,
        config.width,
        config.height
      );
    }
  }
}
```

2. 在 Building 类中集成 RoomRenderer
