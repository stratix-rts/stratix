```typescript
public assignAgent(agentId: string): void {
  // 找到第一个空格子间
  const emptyRoom = this.findEmptyRoom();
  
  if (!emptyRoom) {
    console.warn('[Building] No available room for agent');
    return;
  }
  
  // 分配 Agent 到格子间
  emptyRoom.assignAgent(agentId);
  
  // 更新建筑标签的 Agent 数量
  const occupiedCount = this.countOccupiedRooms();
  this.buildingLabel?.updateAgentCount(occupiedCount);
}

private findEmptyRoom(): Room | undefined {
  for (const room of this.rooms) {
    if (room.getConfig().state === RoomState.Empty) {
      return room;
    }
  }
  return undefined;
}

private countOccupiedRooms(): number {
  return this.rooms.filter(r => r.getConfig().state === RoomState.Occupied).length;
}
```
