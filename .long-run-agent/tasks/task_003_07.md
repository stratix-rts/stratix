## 实现方案

1. 创建 `src/stratix-rts/buildings/BuildingLabel.ts`：
```typescript
export class BuildingLabel extends Phaser.GameObjects.Container {
  private nameText: Phaser.GameObjects.Text;
  private agentCountText: Phaser.GameObjects.Text;
  private progressBar: Phaser.GameObjects.Graphics;
  
  constructor(scene: Phaser.Scene, x: number, y: number, buildingName: string) {
    super(scene, x, y);
    this.createLabel();
    this.updateName(buildingName);
    this.updateAgentCount(0);
    this.updateProgress(0);
  }
  
  private createLabel(): void {
    // 名称
    this.nameText = this.scene.add.text(0, -20, '', {
      fontSize: '14px',
      color: '#ffffff',
      fontStyle: 'bold'
    }).setOrigin(0.5);
    
    // Agent 数量
    this.agentCountText = this.scene.add.text(0, 0, '', {
      fontSize: '12px',
      color: '#ffffff'
    }).setOrigin(0.5);
    
    // 进度条
    this.progressBar = this.scene.add.graphics();
    
    this.add([this.nameText, this.agentCountText, this.progressBar]);
  }
  
  public updateName(name: string): void {
    this.nameText.setText(name);
  }
  
  public updateAgentCount(count: number): void {
    this.agentCountText.setText(`👥 ${count}`);
  }
  
  public updateProgress(percent: number): void {
    this.progressBar.clear();
    
    // 背景
    this.progressBar.fillStyle(0x374151, 1);
    this.progressBar.fillRect(-40, 15, 80, 6);
    
    // 进度
    this.progressBar.fillStyle(0x10b981, 1);
    this.progressBar.fillRect(-40, 15, 80 * percent, 6);
  }
}
```

2. 在 Building 类中集成 BuildingLabel
