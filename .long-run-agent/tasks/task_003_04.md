## 实现方案

1. 创建 `src/stratix-rts/buildings/BuildingRenderer.ts`：
```typescript
export class BuildingRenderer {
  private scene: Phaser.Scene;
  private shellGraphics: Phaser.GameObjects.Graphics;
  
  constructor(scene: Phaser.Scene) {
    this.scene = scene;
    this.shellGraphics = scene.add.graphics();
  }
  
  public renderShell(x: number, y: number, width: number, height: number, buildingType: BuildingType): void {
    const color = this.getBuildingColor(buildingType);
    
    this.shellGraphics.clear();
    this.shellGraphics.fillStyle(color, 1);
    this.shellGraphics.fillRect(x - width/2, y - height/2, width, height);
    
    // 添加边框
    this.shellGraphics.lineStyle(2, 0x000000, 0.5);
    this.shellGraphics.strokeRect(x - width/2, y - height/2, width, height);
  }
  
  private getBuildingColor(buildingType: BuildingType): number {
    switch (buildingType) {
      case BuildingType.DataCenter: return 0x6366f1;  // 紫色
      case BuildingType.Workshop: return 0xf59e0b;  // 橙色
      case BuildingType.Studio: return 0x10b981;  // 绿色
      case BuildingType.Library: return 0x3b82f6;  // 蓝色
      case BuildingType.Command: return 0xef4444;  // 红色
      default: return 0x6b7280;
    }
  }
}
```

2. 在 Building 类中使用 BuildingRenderer
